# The toy is a command now, not a button.
#
# What this proves: the top right of a pro window has no TOY/PRO switch in
# it; /toy still gets you there; and the one button that is in the toy
# brings you back.  The shots are the point - hsk-toy-probar should have
# nothing in that corner, and hsk-toy-in should have one button saying
# BACK TO PRO.
#
geometry 0 0 1400 800
activate
wait 1500
key escape
wait 600
shot /tmp/hsk-toy-probar.png
type /toy
wait 400
key return
wait 1200
shot /tmp/hsk-toy-in.png
# the button lives at the top right; press it
at 1330 26
wait 400
shot /tmp/hsk-toy-hot.png
click l
wait 1200
shot /tmp/hsk-toy-back.png
