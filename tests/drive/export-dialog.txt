# The export dialog: does it open, does it look like the program, and does
# each format bring up its own settings rather than a filter dropdown.
#
# What this proves: Ctrl+E opens a room with the formats down one side, a
# live view of the model in the middle, and options on the right that change
# with the format - and that GIF asks for a recording rather than offering a
# spin of its own.

geometry 0 0 1500 850
activate
wait 600
key escape
key slash
type fit
key return
wait 1500

# Ctrl+E.  hidctl holds a modifier with "hold" and lets it go with "let" -
# it was there all along and I had the syntax wrong.
hold ctrl
key e
let ctrl
wait 1800
shot /tmp/hsk-export-png.png

# JPEG - a quality slider should appear
at 388 226
click
wait 700
shot /tmp/hsk-export-jpeg.png

# GIF - the axes, the loop, and the way in to the recording room.  No
# seconds box and no frame rate box: the recording decides both.
at 388 264
click
wait 700
shot /tmp/hsk-export-gif.png

# STL - no picture settings at all, and the note underneath reads in full
at 388 428
click
wait 700
shot /tmp/hsk-export-stl.png
